-- =========================
-- Inventory Module
-- =========================

CREATE TABLE business (
    id SERIAL PRIMARY KEY,
    name VARCHAR(255),
    cen VARCHAR(64),
    is_active BOOLEAN NOT NULL DEFAULT TRUE
);

CREATE TABLE category (
    id SERIAL PRIMARY KEY,
    business_id INT NOT NULL,
    name VARCHAR(255),
    description VARCHAR(255),
    cen VARCHAR(64),
    is_active BOOLEAN NOT NULL DEFAULT TRUE,

    FOREIGN KEY (business_id) REFERENCES business(id)
);

CREATE TABLE unit (
    id SERIAL PRIMARY KEY,
    business_id INT NOT NULL,
    name VARCHAR(255),
    description VARCHAR(255),
    abbreviation VARCHAR(50),
    cen VARCHAR(64),
    is_active BOOLEAN NOT NULL DEFAULT TRUE,

    FOREIGN KEY (business_id) REFERENCES business(id)
);

CREATE TABLE product_status (
    id SERIAL PRIMARY KEY,
    name VARCHAR(255),
    description VARCHAR(255)
);

-- =========================

CREATE TABLE product (
    id SERIAL PRIMARY KEY,
    business_id INT NOT NULL,
    cen VARCHAR(64),
    sku VARCHAR(100),
    name VARCHAR(255) NOT NULL,
    description VARCHAR(255),
    price NUMERIC(12,2) NOT NULL,
    cost_price NUMERIC(12,2),
    is_active BOOLEAN NOT NULL DEFAULT TRUE,
    unit_id INT NOT NULL,
    unit_qty DOUBLE PRECISION NOT NULL,
    category_id INT NOT NULL,
    station_code VARCHAR(50),
    reorder_level INT NOT NULL DEFAULT 0,

    FOREIGN KEY (business_id) REFERENCES business(id),
    FOREIGN KEY (unit_id) REFERENCES unit(id),
    FOREIGN KEY (category_id) REFERENCES category(id)
);

CREATE TABLE warehouse (
    id SERIAL PRIMARY KEY,
    business_id INT,
    name VARCHAR(255),
    cen VARCHAR(64),
    is_active BOOLEAN NOT NULL DEFAULT TRUE,

    FOREIGN KEY (business_id) REFERENCES business(id)
);

CREATE TABLE warehouse_product (
    id SERIAL PRIMARY KEY,
    warehouse_id INT,
    product_id INT,
    status_id INT,
    price DOUBLE PRECISION NOT NULL,
    stock_left INT,
    low_stock_qty INT,

    FOREIGN KEY (status_id) REFERENCES product_status(id),
    FOREIGN KEY (warehouse_id) REFERENCES warehouse(id),
    FOREIGN KEY (product_id) REFERENCES product(id)
);

CREATE TABLE kardex (
    id SERIAL PRIMARY KEY,
    warehouse_id INT,
    product_id INT,
    document_id INT,
    movement_cen VARCHAR(64),
    action_type VARCHAR(50),
    action_qty DOUBLE PRECISION,
    time_stamp TIMESTAMP,
    reason VARCHAR(255),

    FOREIGN KEY (warehouse_id) REFERENCES warehouse(id),
    FOREIGN KEY (product_id) REFERENCES product(id)
);

CREATE TABLE inventory_document (
    id SERIAL PRIMARY KEY,
    business_id INT NOT NULL,
    warehouse_id INT NOT NULL,
    document_cen VARCHAR(64) NOT NULL,
    document_type VARCHAR(50) NOT NULL,
    status VARCHAR(50) NOT NULL,
    reason VARCHAR(255),
    external_reference VARCHAR(255),
    source VARCHAR(50),
    reference_cen VARCHAR(100),
    created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,

    FOREIGN KEY (business_id) REFERENCES business(id),
    FOREIGN KEY (warehouse_id) REFERENCES warehouse(id)
);

CREATE TABLE inventory_document_line (
    id SERIAL PRIMARY KEY,
    document_id INT NOT NULL,
    product_id INT NOT NULL,
    quantity DOUBLE PRECISION NOT NULL,
    unit_cost NUMERIC(12,2),

    FOREIGN KEY (document_id) REFERENCES inventory_document(id),
    FOREIGN KEY (product_id) REFERENCES product(id)
);

-- =========================
-- Sales Module
-- =========================

CREATE TABLE customer (
    id SERIAL PRIMARY KEY,
    name VARCHAR(255),
    phone VARCHAR(50)
);

CREATE TABLE order_status (
    id SERIAL PRIMARY KEY,
    name VARCHAR(255),
    description VARCHAR(255)
);

CREATE TABLE payment_type (
    id SERIAL PRIMARY KEY,
    name VARCHAR(255),
    description VARCHAR(255)
);

-- =========================

CREATE TABLE order_ticket (
    id SERIAL PRIMARY KEY,
    customer_id INT,
    status_id INT,
    tax_rate_snapshot NUMERIC(7,4) NOT NULL DEFAULT 0,

    FOREIGN KEY (customer_id) REFERENCES customer(id),
    FOREIGN KEY (status_id) REFERENCES order_status(id)
);

CREATE TABLE global_tax_config (
    id SERIAL PRIMARY KEY,
    tax_rate NUMERIC(7,4) NOT NULL
);

CREATE TABLE order_item (
    id SERIAL PRIMARY KEY,
    qty DOUBLE PRECISION,
    additional_note VARCHAR(255),
    order_id INT,
    product_id INT,
    product_cen VARCHAR(64),
    status_id INT,

    FOREIGN KEY (order_id) REFERENCES order_ticket(id),
    FOREIGN KEY (product_id) REFERENCES product(id),
    FOREIGN KEY (status_id) REFERENCES order_status(id)
);

CREATE TABLE payment (
    id SERIAL PRIMARY KEY,
    order_id INT,
    payment_type_id INT,
    paid_at TIMESTAMP,

    FOREIGN KEY (order_id) REFERENCES order_ticket(id),
    FOREIGN KEY (payment_type_id) REFERENCES payment_type(id)
);

-- =========================
-- POS Submodule
-- =========================

CREATE TABLE waiter (
    id SERIAL PRIMARY KEY,
    name VARCHAR(255),
    phone VARCHAR(50)
);

CREATE TABLE station_type (
    id SERIAL PRIMARY KEY,
    name VARCHAR(255),
    description VARCHAR(255)
);

-- =========================

CREATE TABLE order_command (
    id SERIAL PRIMARY KEY,
    order_id INT,
    waiter_id INT,

    FOREIGN KEY (order_id) REFERENCES order_ticket(id),
    FOREIGN KEY (waiter_id) REFERENCES waiter(id)
);

CREATE TABLE station (
    id SERIAL PRIMARY KEY,
    name VARCHAR(255),
    type_id INT,

    FOREIGN KEY (type_id) REFERENCES station_type(id)
);

CREATE TABLE command_item (
    id SERIAL PRIMARY KEY,
    order_item_id INT,
    command_id INT,
    station_id INT,

    FOREIGN KEY (order_item_id) REFERENCES order_item(id),
    FOREIGN KEY (station_id) REFERENCES station(id)
);

CREATE TABLE station_coverage (
    station_type_id INT,
    category_id INT,

    PRIMARY KEY (station_type_id, category_id),

    FOREIGN KEY (station_type_id) REFERENCES station_type(id),
    FOREIGN KEY (category_id) REFERENCES category(id)
);
